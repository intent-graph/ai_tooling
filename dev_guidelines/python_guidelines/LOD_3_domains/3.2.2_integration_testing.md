## Integration Test Scope
<!-- Lines 16-46 -->

Integration tests verify that components work together correctly.

**Unit vs Integration:**
| Aspect | Unit Test | Integration Test |
|--------|-----------|-----------------|
| Scope | Single function/class | Multiple components |
| Dependencies | Mocked | Real (or realistic) |
| Speed | Fast (<100ms) | Slower (seconds) |
| Count | Many | Fewer |

**What to Integration Test:**
- Database operations
- API endpoints
- Multi-service workflows
- File system operations

**Test Pyramid Balance:**
```
        /\           Few E2E tests
       /  \
      /----\         Some integration tests
     /------\
    /--------\       Many unit tests
```

Test database operations with real (test) databases.

**SQLite for Unit Tests:**
```python
import pytest
from sqlalchemy import create_engine
from sqlalchemy.orm import Session

@pytest.fixture
def db_session():
    engine = create_engine("sqlite:///:memory:")
    Base.metadata.create_all(engine)

    with Session(engine) as session:
        yield session
        session.rollback()
```

import pytest
from testcontainers.postgres import PostgresContainer

**Test Database Operations:**
```python
def test_create_user(db_session):
    # Create
    user = User(name="Alice", email="alice@example.com")
    db_session.add(user)
    db_session.commit()

    # Verify
    result = db_session.query(User).filter_by(name="Alice").first()
    assert result is not None
    assert result.email == "alice@example.com"

def test_user_unique_email(db_session):
    user1 = User(name="Alice", email="alice@example.com")
    db_session.add(user1)
    db_session.commit()

    user2 = User(name="Bob", email="alice@example.com")
    db_session.add(user2)

    with pytest.raises(IntegrityError):
        db_session.commit()
```

## API Testing
<!-- Lines 120-182 -->

Test HTTP endpoints with test clients.

**FastAPI Testing:**
```python
from fastapi.testclient import TestClient
from myapp import app

@pytest.fixture
def client():
    return TestClient(app)

def test_create_user(client):
    response = client.post(
        "/users",
        json={"name": "Alice", "email": "alice@example.com"}
    )

    assert response.status_code == 201
    assert response.json()["name"] == "Alice"

def test_get_user_not_found(client):
    response = client.get("/users/999")
    assert response.status_code == 404
```

**Flask Testing:**
```python
import pytest
from myapp import create_app

@pytest.fixture
def client():
    app = create_app(testing=True)
    with app.test_client() as client:
        yield client

def test_index(client):
    response = client.get("/")
    assert response.status_code == 200
```

**Testing Authentication:**
```python
@pytest.fixture
def auth_headers(client):
    response = client.post("/auth/login", json={
        "username": "testuser",
        "password": "testpass"
    })
    token = response.json()["token"]
    return {"Authorization": f"Bearer {token}"}

def test_protected_endpoint(client, auth_headers):
    response = client.get("/protected", headers=auth_headers)
    assert response.status_code == 200
```

## Test Isolation
<!-- Lines 185-248 -->

Ensure tests don't affect each other.

**Database Isolation:**
```python
@pytest.fixture
def db_session():
    # Start transaction
    connection = engine.connect()
    transaction = connection.begin()
    session = Session(bind=connection)

    yield session

    # Rollback everything
    session.close()
    transaction.rollback()
    connection.close()
```

**File System Isolation:**
```python
import tempfile
import pytest

@pytest.fixture
def temp_dir():
    with tempfile.TemporaryDirectory() as tmpdir:
        yield Path(tmpdir)

def test_file_operations(temp_dir):
    file_path = temp_dir / "test.txt"
    file_path.write_text("content")
    assert file_path.read_text() == "content"
```

**Environment Isolation:**
```python
@pytest.fixture
def clean_env(monkeypatch):
    # Remove all environment variables
    for key in list(os.environ.keys()):
        monkeypatch.delenv(key, raising=False)

    # Set only what's needed
    monkeypatch.setenv("APP_ENV", "test")

def test_config_loading(clean_env, monkeypatch):
    monkeypatch.setenv("DATABASE_URL", "sqlite:///:memory:")
    config = load_config()
    assert config.database_url == "sqlite:///:memory:"
```
