## Core Rules (Always Enable)
<!-- Lines 17-24 -->

```toml
select = [
    "E",   # pycodestyle errors
    "F",   # Pyflakes (undefined names, unused imports)
]
```

## Recommended Extensions
<!-- Lines 27-36 -->

```toml
select = [
    "E", "F",   # Core
    "I",        # isort
    "B",        # flake8-bugbear
    "UP",       # pyupgrade
]
```

## Progressive Expansion
<!-- Lines 39-56 -->

```toml
# Level 1: Essential
select = ["E", "F"]

# Level 2: + Style
select = ["E", "F", "W", "I"]

# Level 3: + Bug prevention
select = ["E", "F", "W", "I", "B", "UP"]

# Level 4: + Code quality
select = ["E", "F", "W", "I", "B", "UP", "SIM", "PTH"]

# Level 5: + Security
select = ["E", "F", "W", "I", "B", "UP", "SIM", "PTH", "S"]
```
