> Preserve function signatures in decorators.

## Problem
<!-- Lines 16-24 -->

```python
# Without ParamSpec, signature is lost
def decorator(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper  # Type: (...) -> Any
```

## Solution
<!-- Lines 27-46 -->

```python
from typing import ParamSpec, TypeVar, Callable

P = ParamSpec("P")
R = TypeVar("R")

def logged(func: Callable[P, R]) -> Callable[P, R]:
    def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
        print(f"Calling {func.__name__}")
        return func(*args, **kwargs)
    return wrapper

@logged
def greet(name: str, excited: bool = False) -> str:
    return f"Hello, {name}{'!' if excited else ''}"

# Signature preserved: (name: str, excited: bool = False) -> str
```

## With Return Transform
<!-- Lines 49-56 -->

```python
def returns_list(func: Callable[P, R]) -> Callable[P, list[R]]:
    def wrapper(*args: P.args, **kwargs: P.kwargs) -> list[R]:
        return [func(*args, **kwargs)]
    return wrapper
```
