> Where to patch for effective mocking.

## Rule: Patch Where Used
<!-- Lines 17-37 -->

```python
# myapp/service.py
from myapp.client import api_client

def get_data():
    return api_client.fetch()

# test_service.py
# CORRECT: Patch where it's used
@patch("myapp.service.api_client")
def test_get_data(mock_client):
    mock_client.fetch.return_value = {"key": "value"}
    result = get_data()

# WRONG: Patching the source
@patch("myapp.client.api_client")  # Won't work!
def test_get_data(mock_client):
    ...
```

## Patch Object Attribute
<!-- Lines 40-46 -->

```python
@patch.object(UserService, "fetch_user")
def test_service(mock_fetch):
    mock_fetch.return_value = User(id=1)
```

## Patch Multiple
<!-- Lines 49-56 -->

```python
@patch("myapp.service.cache")
@patch("myapp.service.database")
def test_service(mock_db, mock_cache):  # Reverse order!
    ...
```

## Context Manager
<!-- Lines 59-66 -->

```python
def test_service():
    with patch("myapp.service.api_client") as mock:
        mock.fetch.return_value = {}
        result = get_data()
```
