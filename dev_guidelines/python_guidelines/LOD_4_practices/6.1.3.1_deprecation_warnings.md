> Emitting and handling deprecation warnings in Python.

## Basic Warning
<!-- Lines 21-33 -->

```python
import warnings

def old_function(x):
    warnings.warn(
        "old_function is deprecated, use new_function instead",
        DeprecationWarning,
        stacklevel=2  # Point to caller
    )
    return new_function(x)
```

## stacklevel Explained
<!-- Lines 36-52 -->

```python
def deprecated_func():        # stacklevel=1 (this line)
    warnings.warn(            # stacklevel=2 (caller)
        "deprecated",
        DeprecationWarning,
        stacklevel=2
    )

# For wrapped functions, increase stacklevel
def wrapper():               # stacklevel=2
    deprecated_func()        # stacklevel=1

# Call from user code
wrapper()                    # stacklevel=3
```

## Warning Categories
<!-- Lines 55-66 -->

```python
# For developer/library users (hidden by default)
warnings.warn("...", DeprecationWarning)

# For upcoming deprecation
warnings.warn("...", PendingDeprecationWarning)

# For end users (visible by default)
warnings.warn("...", FutureWarning)
```

## Decorator Pattern
<!-- Lines 69-92 -->

```python
import functools
import warnings

def deprecated(reason: str, version: str = None):
    def decorator(func):
        msg = f"{func.__name__} is deprecated"
        if version:
            msg += f" since {version}"
        msg += f". {reason}"

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            warnings.warn(msg, DeprecationWarning, stacklevel=2)
            return func(*args, **kwargs)
        return wrapper
    return decorator

@deprecated("Use new_api() instead", version="2.0")
def old_api():
    pass
```

## Filtering Warnings
<!-- Lines 110-129 -->

```python
# In tests: Treat deprecations as errors
import warnings
warnings.filterwarnings("error", category=DeprecationWarning)

# Ignore specific deprecation
warnings.filterwarnings(
    "ignore",
    message="old_function.*",
    category=DeprecationWarning,
    module="mypackage"
)

# Via pytest.ini
# filterwarnings =
#     error::DeprecationWarning
#     ignore::DeprecationWarning:thirdparty
```

## Testing Deprecation Warnings
<!-- Lines 132-146 -->

```python
import pytest
import warnings

def test_deprecation_warning_emitted():
    with pytest.warns(DeprecationWarning, match="old_function"):
        old_function()

def test_no_warning_for_new_function():
    with warnings.catch_warnings():
        warnings.simplefilter("error")
        new_function()  # Should not warn
```
