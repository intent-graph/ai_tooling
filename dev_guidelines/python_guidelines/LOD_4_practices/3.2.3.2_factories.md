def create_user(**overrides) -> User:
    defaults = {
        "name": "Test User",
        "email": "test@example.com",
    }
    return User(**{**defaults, **overrides})

# Usage
user = create_user(name="Alice")

## Factory Class
<!-- Lines 33-49 -->

```python
class UserFactory:
    _counter = 0

    @classmethod
    def create(cls, **overrides) -> User:
        cls._counter += 1
        defaults = {
            "name": f"User {cls._counter}",
            "email": f"user{cls._counter}@example.com",
        }
        return User(**{**defaults, **overrides})

user1 = UserFactory.create()  # Unique each time
```

## factory_boy
<!-- Lines 52-69 -->

```python
import factory

class UserFactory(factory.Factory):
    class Meta:
        model = User

    name = factory.Faker("name")
    email = factory.LazyAttribute(
        lambda o: f"{o.name.lower().replace(' ', '.')}@example.com"
    )

# Usage
user = UserFactory()
users = UserFactory.create_batch(5)
```
