## Full Config
<!-- Lines 26-50 -->

```toml
[tool.mypy]
python_version = "3.10"

# Strictness
strict = true
warn_return_any = true
warn_unused_ignores = true
warn_redundant_casts = true
disallow_any_generics = true

# Output
show_error_codes = true
show_column_numbers = true
pretty = true

# Performance
incremental = true
cache_dir = ".mypy_cache"

# Plugins
plugins = ["pydantic.mypy"]
```

<!-- Lines 70-76 -->

```bash

```
