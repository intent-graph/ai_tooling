## Basic Usage
<!-- Lines 18-29 -->

```python
from multiprocessing import Pool

def cpu_intensive(n):
    return sum(i * i for i in range(n))

if __name__ == "__main__":
    with Pool(processes=4) as pool:
        results = pool.map(cpu_intensive, [10**6] * 10)
```

## Pool Methods
<!-- Lines 32-50 -->

```python
with Pool(4) as pool:
    # Blocking map
    results = pool.map(func, items)

    # Non-blocking
    async_result = pool.map_async(func, items)
    results = async_result.get()

    # Iterator (memory-efficient)
    for result in pool.imap(func, items):
        process(result)

    # Unordered (fastest)
    for result in pool.imap_unordered(func, items):
        process(result)
```

## Single Process Call
<!-- Lines 53-68 -->

```python
from multiprocessing import Process, Queue

def worker(q, data):
    result = compute(data)
    q.put(result)

if __name__ == "__main__":
    q = Queue()
    p = Process(target=worker, args=(q, data))
    p.start()
    result = q.get()
    p.join()
```

## Shared Memory (Large Data)
<!-- Lines 71-85 -->

```python
from multiprocessing import shared_memory
import numpy as np

# Create shared array
shm = shared_memory.SharedMemory(create=True, size=arr.nbytes)
shared_arr = np.ndarray(arr.shape, dtype=arr.dtype, buffer=shm.buf)
np.copyto(shared_arr, arr)

# In worker: access by name
existing_shm = shared_memory.SharedMemory(name=shm.name)
arr = np.ndarray(shape, dtype=dtype, buffer=existing_shm.buf)
```
