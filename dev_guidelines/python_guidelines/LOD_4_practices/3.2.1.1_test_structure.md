> Arrange, Act, Assert pattern.

## AAA Pattern
<!-- Lines 16-28 -->

```python
def test_user_full_name():
    # Arrange
    user = User(first_name="Alice", last_name="Smith")

    # Act
    result = user.full_name()

    # Assert
    assert result == "Alice Smith"
```

## Complex Arrange
<!-- Lines 31-46 -->

```python
def test_order_total_with_discount():
    # Arrange
    item1 = Item(name="Widget", price=Decimal("10.00"))
    item2 = Item(name="Gadget", price=Decimal("20.00"))
    discount = Discount(percent=10)
    order = Order(items=[item1, item2], discount=discount)

    # Act
    total = order.calculate_total()

    # Assert
    assert total == Decimal("27.00")
```

## Multiple Assertions
<!-- Lines 49-64 -->

```python
# OK: Related assertions
def test_user_creation():
    user = User(name="Alice", age=30)

    assert user.name == "Alice"
    assert user.age == 30
    assert user.is_active  # Default

# Better: Separate test
def test_user_defaults_to_active():
    user = User(name="Alice", age=30)
    assert user.is_active
```
